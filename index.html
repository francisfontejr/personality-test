function calculateResult() {


  let scores = {

    A: 0,

    B: 0,

    C: 0,

    D: 0

  };


  for (let i = 1; i <= 10; i++) {


    let answer = document.querySelector(

      'input[name="q' + i + '"]:checked'

    );


    if (!answer) {

      alert("Please answer all 10 questions.");

      return;

    }


    scores[answer.value]++;

  }



  let maxScore = Math.max(

    scores.A,

    scores.B,

    scores.C,

    scores.D

  );



  let winners = [];


  for (let type in scores) {

    if (scores[type] === maxScore) {

      winners.push(type);

    }

  }



  const names = {

    A: "Introvert",

    B: "Extrovert",

    C: "Ambivert",

    D: "Otrovert"

  };



  const descriptions = {


    A:

    "You tend to prefer deeper one-on-one connections, smaller social circles, and enough alone time to recharge.",


    B:

    "You tend to gain energy from social interaction and enjoy meeting, talking with, and spending time with other people.",


    C:

    "You have a balance of introverted and extroverted tendencies. Your social energy often depends on your mood and situation.",


    D:

    "You can enjoy social interaction while remaining highly independent. You may care less about fitting into groups and prefer meaningful individual connections."


  };



  let title = "";

  let description = "";



  if (winners.length === 1) {


    let winner = winners[0];


    title = "You are mostly an " + names[winner] + "!";


    description = descriptions[winner];


  } else {


    let winnerNames = winners.map(

      winner => names[winner]

    );


    title = "You have a mixed personality!";


    description =

      "Your strongest traits are: " +

      winnerNames.join(" + ") +

      ". You may show different personality traits depending on the situation.";


  }



  document.getElementById("resultTitle").innerText = title;


  document.getElementById("resultDescription").innerText =

    description;


  document.getElementById("scoreDisplay").innerHTML =

    "<strong>Your Scores</strong><br><br>" +

    "Introvert: " + scores.A + "<br>" +

    "Extrovert: " + scores.B + "<br>" +

    "Ambivert: " + scores.C + "<br>" +

    "Otrovert: " + scores.D;



  document.getElementById("result").style.display = "block";



  document.getElementById("result").scrollIntoView({

    behavior: "smooth"

  });


}



function resetQuiz() {


  document.getElementById("quizForm").reset();


  document.getElementById("result").style.display = "none";


  window.scrollTo({

    top: 0,

    behavior: "smooth"

  });


}
